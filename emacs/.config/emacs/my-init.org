** process and GC tuning
Both values come from [[https://github.com/emacs-lsp/lsp-mode#performance][lsp-mode's performance guide]]. A large
=gc-cons-threshold= means fewer full collections during the heavy JSON churn an LSP
session generates; measured cost is low (32 collections totalling 0.9s across a
22-hour session). If stalls ever do show up, reach for =gcmh= -- which adapts the
threshold to whether you are typing or idle -- rather than a smaller constant.
#+begin_src emacs-lisp :tangle "early-init.el" :mkdirp yes
(setq gc-cons-threshold 200000000)        ; 200mb
(setq read-process-output-max (* 10 1024 1024)) ; 10mb
#+end_src

** disable bars before frame creation
Turning the bars off with frame parameters means the first frame is created without them. Calling =menu-bar-mode= or =scroll-bar-mode= from =init.el= would remove them from a frame that is already drawn, which forces a resize and redraw. The mode variables are set to match so toggling them later still works.
#+begin_src emacs-lisp :tangle "early-init.el" :mkdirp yes
(push '(menu-bar-lines . 0) default-frame-alist)
(push '(tool-bar-lines . 0) default-frame-alist)
(push '(vertical-scroll-bars) default-frame-alist)
(setq menu-bar-mode nil
      tool-bar-mode nil
      scroll-bar-mode nil)
(setq frame-inhibit-implied-resize t)
#+end_src

** paint the first frame in the theme's colors
The theme loads only after elpaca has activated every package, so the first frame would otherwise be drawn white and opaque until then. These parameters give the first frame the doom-one background and foreground and the transparency from =my/alpha-background= straight away; keep them in step with those.

They go in =initial-frame-alist=, not =default-frame-alist=: color parameters in =default-frame-alist= would override the theme on every later frame, including =emacsclient= frames.
#+begin_src emacs-lisp :tangle "early-init.el" :mkdirp yes
(push '(background-color . "#282c34") initial-frame-alist)
(push '(foreground-color . "#bbc2cf") initial-frame-alist)
(push (cons (if (eq system-type 'darwin) 'alpha 'alpha-background) 90)
      initial-frame-alist)
(setq inhibit-startup-screen t
      inhibit-startup-echo-area-message user-login-name)
#+end_src

** start maximized
Asking for this in =initial-frame-alist= means the very first frame is created
maximized. The old =window-setup-hook= + =toggle-frame-maximized= approach drew a
small frame first and then resized it, which was visible on every launch.

It must not go in =default-frame-alist=: child frames (lsp-ui-doc, posframes)
inherit that, and on macOS a maximized child frame ignores resizing and stays one
character cell in size.
#+begin_src emacs-lisp :tangle "early-init.el" :mkdirp yes
(add-to-list 'initial-frame-alist '(fullscreen . maximized))
#+end_src

** silence harmless async native-comp warnings
Third-party packages (elpaca builds) routinely trip "not known to be
defined" warnings for functions guarded by fboundp/version checks, or
forward-referenced later in the same file. These are compile-time
noise, not runtime bugs, and popping up a warnings buffer for them on
every async compile is more distracting than useful.
#+begin_src emacs-lisp :tangle "early-init.el" :mkdirp yes
(setq native-comp-async-report-warnings-errors 'silent)
#+end_src

** Find gcc and libgccjit packages to be used for emacs on macOS
#+begin_src emacs-lisp :tangle "early-init.el" :mkdirp yes
(defun homebrew-gcc-paths ()
  "Return GCC library paths from Homebrew installations.
  Detects paths for gcc and libgccjit packages to be used in LIBRARY_PATH."
  (let* ((paths '())
         (brew-bin
          (or (executable-find "brew")
              (let ((arm-path "/opt/homebrew/bin/brew")
                    (intel-path "/usr/local/bin/brew"))
                (cond
                 ((file-exists-p arm-path)
                  arm-path)
                 ((file-exists-p intel-path)
                  intel-path))))))

    (when brew-bin
      ;; Get gcc paths.
      (let* ((gcc-prefix
              (string-trim
               (shell-command-to-string
                (concat brew-bin " --prefix gcc"))))
             (gcc-lib-current
              (expand-file-name "lib/gcc/current" gcc-prefix)))
        (push gcc-lib-current paths)

        ;; Find apple-darwin directory.
        (let* ((default-directory gcc-lib-current)
               (arch-dirs
                (file-expand-wildcards "gcc/*-apple-darwin*/*[0-9]")))
          (when arch-dirs
            (push (expand-file-name (car (sort arch-dirs #'string>)))
                  paths))))

      ;; Get libgccjit paths
      (let* ((jit-prefix
              (string-trim
               (shell-command-to-string
                (concat brew-bin " --prefix libgccjit"))))
             (jit-lib-current
              (expand-file-name "lib/gcc/current" jit-prefix)))
        (push jit-lib-current paths)))

    (nreverse paths)))

(defun setup-macos-native-comp-library-paths ()
  "Set up LIBRARY_PATH for native compilation on macOS.
  Includes Homebrew GCC paths and CommandLineTools SDK libraries."
  (let*
      ((existing-paths
        (split-string (or (getenv "LIBRARY_PATH") "") ":" t))
       (gcc-paths (homebrew-gcc-paths))
       (clt-paths
        '("/Library/Developer/CommandLineTools/SDKs/MacOSX.sdk/usr/lib"))
       (unique-paths
        (delete-dups (append existing-paths gcc-paths clt-paths))))

    (setenv "LIBRARY_PATH" (mapconcat #'identity unique-paths ":"))))

;; Set up library paths for native compilation on macOS.
(when (eq system-type 'darwin)
  (setup-macos-native-comp-library-paths))
#+end_src

This is the standard install code block for elpaca

#+begin_src emacs-lisp :tangle "init.el" :mkdirp yes
;;; init.el --- Main init: elpaca bootstrap and module loading  -*- lexical-binding: t; -*-
(defvar elpaca-installer-version 0.12)
(defvar elpaca-directory (expand-file-name "elpaca/" user-emacs-directory))
(defvar elpaca-builds-directory (expand-file-name "builds/" elpaca-directory))
(defvar elpaca-sources-directory (expand-file-name "sources/" elpaca-directory))
(defvar elpaca-order '(elpaca :repo "https://github.com/progfolio/elpaca.git"
                              :ref nil :depth 1 :inherit ignore
                              :files (:defaults "elpaca-test.el" (:exclude "extensions"))
                              :build (:not elpaca-activate)))
(let* ((repo  (expand-file-name "elpaca/" elpaca-sources-directory))
       (build (expand-file-name "elpaca/" elpaca-builds-directory))
       (order (cdr elpaca-order))
       (default-directory repo))
  (add-to-list 'load-path (if (file-exists-p build) build repo))
  (unless (file-exists-p repo)
    (make-directory repo t)
    (when (<= emacs-major-version 28) (require 'subr-x))
    (condition-case-unless-debug err
        (if-let* ((buffer (pop-to-buffer-same-window "*elpaca-bootstrap*"))
                  ((zerop (apply #'call-process `("git" nil ,buffer t "clone"
                                                  ,@(when-let* ((depth (plist-get order :depth)))
                                                      (list (format "--depth=%d" depth) "--no-single-branch"))
                                                  ,(plist-get order :repo) ,repo))))
                  ((zerop (call-process "git" nil buffer t "checkout"
                                        (or (plist-get order :ref) "--"))))
                  (emacs (concat invocation-directory invocation-name))
                  ((zerop (call-process emacs nil buffer nil "-Q" "-L" "." "--batch"
                                        "--eval" "(byte-recompile-directory \".\" 0 'force)")))
                  ((require 'elpaca))
                  ((elpaca-generate-autoloads "elpaca" repo)))
            (progn (message "%s" (buffer-string)) (kill-buffer buffer))
          (error "%s" (with-current-buffer buffer (buffer-string))))
      ((error) (warn "%s" err) (delete-directory repo 'recursive))))
  (unless (require 'elpaca-autoloads nil t)
    (require 'elpaca)
    (elpaca-generate-autoloads "elpaca" repo)
    (let ((load-source-file-function nil)) (load "./elpaca-autoloads"))))
(add-hook 'after-init-hook #'elpaca-process-queues)
(elpaca `(,@elpaca-order))
#+end_src

*** Org from the upstream development branch
Replaces the Org bundled with Emacs. Must stay above every package that requires =org=, and must be followed by =elpaca-wait= so the build is on =load-path= before they load.

Cloned from =bzg/org-mode=, the GitHub mirror of the Savannah repository kept by Bastien Guerry (long-time Org maintainer; the repository is marked mirror-only), because =git.savannah.gnu.org= is blocked or unreliable on some networks. Org's own docs also list =https://git.sr.ht/~bzg/org-mode= as a non-GitHub mirror. Keep =:branch "main"=: the =emacsmirror/org= mirror defaults to =master= and lags the development branch.

The development branch is only here for =org-latex-preview=. Once the Org bundled with Emacs ships that library, a startup warning says this block can go. The check looks in Emacs's own =lisp/org/= directory rather than at =load-path=, because the elpaca build shadows the bundled Org there.
#+begin_src emacs-lisp :tangle "init.el" :mkdirp yes
(use-package org
  :ensure (org :host github :repo "bzg/org-mode" :branch "main")
  :defer t)
(elpaca-wait)

(when (locate-file "org-latex-preview"
                   (list (expand-file-name "org" lisp-directory))
                   '(".elc" ".el" ".el.gz"))
  (display-warning
   'tony-init
   (format "Emacs %s bundles org-latex-preview; remove the Org development branch block from my-init.org."
           emacs-version)))
#+end_src

(defun enable-hyper-super-modifiers-linux-x ()
  "Make the <menu> key act as the hyper modifier.
On modern Linux the <windows> key is usually already mapped to Super.
Note: for H-s you must release <menu> before pressing `s'."
  (define-key key-translation-map [menu] 'event-apply-hyper-modifier))

;; Only meaningful under X -- there is no <menu> key event to translate in a
;; terminal frame, and macOS has its own modifier handling.
(defun tony/enable-hyper-in-x-frame (frame)
  "Install the <menu> hyper translation on FRAME's terminal if it is X."
  (when (eq (window-system frame) 'x)
    (with-selected-frame frame
      (enable-hyper-super-modifiers-linux-x))))

(when (eq window-system 'x)
  (enable-hyper-super-modifiers-linux-x))
(add-hook 'after-make-frame-functions #'tony/enable-hyper-in-x-frame)

;; Vertico settings
;; Add prompt indicator to `completing-read-multiple'.
;; We display [CRM<separator>], e.g., [CRM,] if the separator is a comma.
(require 'crm) ; for `crm-separator'
(defun crm-indicator (args)
  (cons
   (format "[CRM%s] %s"
           (replace-regexp-in-string
            "\\`\\[.*?]\\*\\|\\[.*?]\\*\\'" "" crm-separator)
           (car args))
   (cdr args)))
(advice-add #'completing-read-multiple :filter-args #'crm-indicator)

**** gdshader-mode + gdshader-lsp-cpp (DIY shader language server)
Godot shader files (=.gdshader=, =.gdshaderinc=). Godot's built-in language server only
serves GDScript, so shaders need two separate pieces, one of which is built by hand.

***** the pieces
- =gdshader-mode= :: regex-based major mode from MELPA ([[https://github.com/bbbscarter/gdshader-mode][bbbscarter/gdshader-mode]]).
  Builds on =glsl-mode= (installed as a dependency) and adds Godot's extensions.
  Gives highlighting and indentation only. There is no tree-sitter mode for it, so
  =treesit-fold= (and the =z= fold key) does nothing in shader buffers.
- =gdshader-lsp-cpp= :: language server, [[https://github.com/Dead-Shrimp-Studio/gdshader-lsp-cpp][Dead-Shrimp-Studio/gdshader-lsp-cpp]]
  (C++, MIT). *Not packaged anywhere: there is no prebuilt binary and Elpaca does not
  manage it.* It lives in =~/.local/src/gdshader-lsp-cpp= and is built from source.
  Capabilities were checked by hand (v0.3.1, request/reply over stdio against a scratch
  shader with a struct, a function, a uniform, a const and an =#include=):
  - Works: completion (built-ins, user and included symbols, struct members, swizzles
    after =.=), signature help (user and included functions), hover (user functions,
    uniforms, consts and built-ins such as =UV=), document symbols, semantic tokens.
  - Partial: go to definition works for variables, uniforms, consts and functions in
    the same file. It returns nothing for struct type names, struct members and the
    path in =#include "..."=. For a function defined in an included file the line range
    is right but the URI is the /including/ file, so the jump lands in the wrong file.
  - Hover gives nothing for type keywords such as =vec4= (empty reply): no docs for
    types or constructors, despite the README's "documentation for symbols".
  - The server does not advertise hover, definition, signature help or document
    symbols in its =initialize= reply even though it implements them, and lsp-mode only
    sends requests for advertised features. The =:initialized-fn= in the client entry
    below adds those capabilities itself. Advertised and used as-is: completion,
    rename, references, folding ranges, inlay hints, semantic tokens, code actions,
    call hierarchy, color provider, workspace symbols (rename/references/inlay
    hints/code actions/etc. not tested).
- Release tracks follow the Godot version: =v0.2.x= (branch =support/4.5+=) for Godot
  4.5/4.6, =v0.3.x= (=master=) for Godot 4.7+. Built from =master= for Godot 4.7.2.

***** machine-specific behaviour
The LSP is only set up where the binary exists (=file-executable-p= guard in the
block below), so machines where it was never built (e.g. the work macOS Emacs) are not
affected: they just get highlighting from =gdshader-mode=, with no failed-start noise.
The binary path is Linux-specific (=bin/linux/release/...=).

***** how to (re)build it
Needs a C++20 compiler (g++ or clang++), CMake and SCons. On Artix/Arch:
#+begin_example
sudo pacman -S cmake scons
#+end_example
(Fallback if there is no sudo: =python3 -m venv= a throwaway venv, =pip install cmake
scons= into it, and put its =bin= first on =PATH= for the build. The built binary does
not depend on the venv. The first build here used that; the current binary was
rebuilt cleanly with the pacman packages, cmake 4.4.5 and scons 4.11.1.)
Then:
#+begin_example
git clone --recurse-submodules https://github.com/Dead-Shrimp-Studio/gdshader-lsp-cpp \
    ~/.local/src/gdshader-lsp-cpp
cd ~/.local/src/gdshader-lsp-cpp
(cd extern/lsp-framework && cmake -B build_linux -DCMAKE_BUILD_TYPE=Release \
    && cmake --build build_linux -j"$(nproc)")
scons platform=linux target=release -j"$(nproc)"
#+end_example
Result: =bin/linux/release/gdshader_lsp_release_linux=. Many compiler warnings
(=-Wunused-parameter=) during the build are normal; check the exit status instead.

***** how to update it
After a Godot upgrade or just to pick up fixes: =git pull --recurse-submodules= in
=~/.local/src/gdshader-lsp-cpp=, check the README for which branch matches the Godot
version (see release tracks above), re-run the =scons= line (re-run the =cmake= step
only if =extern/lsp-framework= changed), then restart the server in Emacs with
=M-x lsp-workspace-restart= in a shader buffer. Nothing in this config pins a version.

***** how to check that it works
Standalone, without Emacs (sends an LSP =initialize= request and prints the reply;
should show =serverInfo= =gdshader-lsp= and a =capabilities= object):
#+begin_example
body='{"jsonrpc":"2.0","id":1,"method":"initialize","params":{"processId":null,"rootUri":"file://'"$PWD"'","capabilities":{}}}'
printf 'Content-Length: %d\r\n\r\n%s' ${#body} "$body" \
  | timeout 5 ~/.local/src/gdshader-lsp-cpp/bin/linux/release/gdshader_lsp_release_linux --stdio
#+end_example
In Emacs: open a =.gdshader= file, then =C-h m= should list =gdshader-mode= and
=lsp-mode=; =M-x lsp-describe-session= should show the =gdshader-lsp= server.

***** known bug: go-to-definition lands in the wrong file for =#include= symbols
Found by testing the server directly (v0.3.1, commit =bb49279= of =master=), not yet
seen in day-to-day use. Not reported upstream. No patch applied; the plan is to patch
the local clone once the bug is actually hit.
- When you will hit it: go-to-definition (=C-c l g g=, or =xref= =M-.=) on a function,
  struct, const or variable that comes from a file pulled in with =#include=
  (e.g. a =.gdshaderinc= in the project). Symbols defined in the same file are fine.
  If a project never uses =#include=, this never shows up.
- What it looks like: no error. Emacs jumps to the correct /line numbers/ but in the
  /current/ file, i.e. you land on unrelated code near the top of the shader you are
  editing instead of in the included file. In the test, a =helper()= defined at lines
  1-3 of =inc.gdshaderinc= sent the cursor to lines 1-3 of the including shader.
- Cause: the definition handler in =src/server/gdshader_server.cpp= (the
  =TextDocument_Definition= handler, around lines 453-513) sets
  =loc.uri = params.textDocument.uri= in every branch, i.e. the file the request came
  from, and never the file the symbol was defined in. The needed data exists:
  =Symbol= (=src/gdshader/semantics/symbol.hpp=) already has a =source_path= field that
  the include handling fills in when it imports symbols.
- Planned local fix: in each branch that returns a symbol, build the URI from
  =sym->source_path= when it is non-empty (and not ="builtin"=), falling back to the
  request URI. Then rebuild (=scons platform=linux target=release= in
  =~/.local/src/gdshader-lsp-cpp=) and =M-x lsp-workspace-restart=. Note that a later
  =git pull= may conflict with the local edit; keep it as a commit or a saved diff.
- Related gaps (server returns nothing, Emacs shows "Not found for: ..."): go-to-definition
  on struct type names, on struct members, and on the path inside =#include "..."=. The
  handler only handles identifier, function-call and variable-declaration nodes;
  =IncludeNode= has a =resolvedPath= field it never looks at.

***** known gap: hover is empty for type constructors and declaration names
Same status as the definition bug above: found by testing the server directly (v0.3.1,
commit =bb49279=), not reported upstream, no patch. Lower priority: it is missing
data/handler coverage, not a wrong answer.
- What you see: hover on =vec2=, =vec4=, =mat2= etc. (constructor calls) shows nothing,
  and so does hover on the /name/ in a declaration such as =col= in =vec4 col = ...=
  (hover on later uses of =col= works and shows its type).
- What does work: user and built-in functions (=mix=, =sin= show a return type and a
  description), uniforms, consts, built-in variables (=UV=, =COLOR=). A few built-in
  overloads have no description text in the server's data (=max=: return type only).
- Cause, constructors: the hover handler's function-call branch only replies when
  =lookupFunctions(name)= finds a symbol. Built-in functions come from the generated table
  =src/generated/builtins_data.cpp=, which has no entries for =vec/ivec/uvec/bvec/mat=
  constructors (grep found zero). The type checker treats them separately
  (=validateConstructor= in =semantics/visitors/type_checking_visitor.cpp=, around line
  489: a known type name is validated and the function lookup is skipped), so they are
  validated but never recorded as something hover can describe.
- Cause, declaration names: the hover handler has no branch for variable-declaration
  nodes (it handles identifier, function-call, member-access and type nodes).
- Fixing it means adding descriptions/overloads for the constructors (or a type-hover
  branch that prints them) and a declaration branch; more work than the one-line
  definition fix.

***** things that will bite
- It *must* be started with =--stdio=. Without it the server defaults to TCP mode and
  a client that spawns it silently fails to attach (README warning). The command below
  already passes it.
- The server writes a log file named =gdshader_lsp_log.txt= into its working directory
  (seen when run by hand), so stray copies may appear wherever Emacs was started. The
  README/changelog also mention a =--log-path= option; not verified, not used here.
- =lsp-enable-folding= is =nil= in the lsp-mode block, so the server's folding ranges
  are not used even though it offers them.
- =lsp-language-id-configuration= has to map =gdshader-mode= to ="gdshader"= for
  =lsp-activate-on= to match, which the block does.
- If the first shader opened after startup gets no LSP but the second does, the
  registration is happening too late (it runs when =gdshader-mode= first loads).
- Removing it: delete the =defconst= and the =when= form in the block below, and
  delete =~/.local/src/gdshader-lsp-cpp= (and =cmake=/=scons= if nothing else uses them).

*** org latex math preview
Inline math renders as images through Org's =org-latex-preview=. Fragments must be delimited, e.g. =\(\sum_{i=1}^{3} k_i\)= or =$k_i$=. =org-latex-preview-mode= reveals the source while the cursor is inside a fragment and previews new =\(...\)=, =\[...\]= and =\begin= fragments when the cursor leaves them. =$...$= and =$$...$$= fragments are only previewed by =M-x org-latex-preview=, so write math with the backslash delimiters. Images are sized relative to the surrounding font, so they follow =text-scale-adjust= without any =:zoom= or =:scale= setting; set =:zoom= in =org-latex-preview-appearance-options= only to make math larger or smaller than the text.

| Requirement      | Purpose                                                   |
|------------------+-----------------------------------------------------------|
| =latex=          | compiles each batch of fragments to DVI                   |
| =dvisvgm= (3.2+) | converts DVI to SVG, the default when it is on =PATH=     |
| =preview.sty=    | crops each fragment; without it no previews render        |
| =mylatexformat=  | precompiles the preamble; without it every run recompiles |

On macOS, MacTeX ships all four:

#+begin_src sh
brew install --cask mactex-no-gui
#+end_src

or, with the smaller BasicTeX:

#+begin_src sh
brew install --cask basictex
sudo tlmgr update --self
sudo tlmgr install dvisvgm preview mylatexformat
#+end_src

On Arch Linux, =preview= and =mylatexformat= are in =texlive-latexextra= and =dvisvgm= is packaged on its own:

#+begin_src sh
sudo pacman -S texlive-basic texlive-bin texlive-latex texlive-latexextra dvisvgm
#+end_src

Check with =kpsewhich preview.sty mylatexformat.ltx= (both paths should print) and =dvisvgm --version=.

Org skips the colour commands for a fragment whose neighbour in the buffer has the same colours, assuming the neighbour set them. That neighbour may be a cached image from an earlier run, so a batch of newly compiled fragments can contain no colour command at all, and every image in it comes out black: invisible on a dark theme. =org-latex-preview--tex-styled= is advised to always emit the colour. Previews generated before the advice stay black until =M-x org-latex-preview-clear-cache= and a fresh preview. Check that a regenerated preview adapts to the theme by switching themes: its SVG uses =currentColor=.
#+begin_src emacs-lisp :tangle "tony-emacs-modules/tony-emacs-org.el" :mkdirp yes

(with-eval-after-load 'org-latex-preview
  (define-advice org-latex-preview--tex-styled (:filter-args (args) always-emit-color)
    (pcase-let ((`(,type ,value ,options) args))
      (list type value (plist-put (copy-sequence options) :continue-color nil)))))

*** org math input (laas)
=laas= (LaTeX Auto Activating Snippets) expands keys as they are typed, but only inside math fragments and LaTeX environments, so prose is never touched. =M-x laas-mode= toggles it in the current buffer.

laas requires =texmathp=, which ships with AUCTeX. AUCTeX is not on MELPA and the GNU ELPA menu and Savannah are unavailable here, so =auctex= comes from the GitHub mirror and links only =texmathp.el=. =tempel= gives fraction snippets fields: =M-}= moves from numerator to denominator.

| Keys                     | Result                                          |
|--------------------------+-------------------------------------------------|
| =sin= =cos= =tan=        | =\sin= etc., with a space after                 |
| =;a= =;p= =;q=           | =\alpha= =\pi= =\theta= (=;= prefixes Greek)    |
| =//=                     | =\frac{}{}=                                     |
| =3/=, =(a+b)/=           | =\frac{3}{}=, wrapping the object to the left   |
| =x1=, =sr=, =cb=         | =x_1=, =^2=, =^3=                               |
| ='q= ='.= ='-= ='>=      | wrap the previous object in sqrt, dot, bar, vec |
| ~<=~ ~>=~ ~!=~ ~~=~ ~=>~ | =\leq= =\geq= =\neq= =\approx= =\implies=       |
| =**=, =xx=               | =\cdot=, =\times=                               |
| =;;m=, =;;M=             | =\(\)= or =\[\]= with point between, in prose   |
| =flr=, =cei=             | =\lfloor \rfloor=, =\lceil \rceil= with point between |

=;;m= (inline) and =;;M= (display) start a fragment in prose, so the other keys work straight away: laas only sees math inside a complete fragment, and electric-pair does not close =\(= or =$= correctly. They sit on the =;;= prefix because =;m= and =;d= are already laas snippets, and a key that is a complete snippet cannot also begin a longer one. They only fire at the start of a word, and never inside math, source or example blocks, or verbatim and code markup.

=TAB= at the end of a fragment steps out of it and adds a space, as in Castel's setup, unless a space or punctuation (=-= included) already follows. At the end of a line the space is added too. If the next key you type is punctuation (=. , ? ! : ; )=), the space is removed again. This only covers fragments started with =;;m= and =;;M=, and only when you leave with =TAB=: leaving with the cursor keys adds nothing.

The commands laas expands to but plain LaTeX lacks are defined once, in =my/latex-math-macros=. The preview preamble uses it, and so do the Inkscape figure renderer and the shortcut manager's pre-rendered text (see =inkscape.el=), so a macro added there works in all three.

Two surprises: ~==~ becomes ~&=~ (meant for align environments), and function keys fire mid-word, so =arcsin= becomes =arc\sin=. A backslash before the key suppresses expansion, so type =\arcsin= directly. Word keys such as =sum=, =set= and =tt= go through =my/laas-word-start-p= instead, so they fire only at the start of a word and never inside a hand-typed =\emptyset= or =\mathtt=.

A key that is a complete snippet cannot also begin a longer one, so =lim= rules out =limsup= (use =lsup=) and laas's =nn= subscript rules out =nnn= (use =bcap=). When two keys end on the same keystroke the longer one wins, so =dint= and =<->= are safe next to =int= and =->=.

** ~inkscape.el~
Figures drawn in Inkscape, shown in Org notes with their labels typeset by LaTeX. The workflow follows Gilles Castel's ([[https://castel.dev/post/lecture-notes-2][How I draw figures for my mathematical lecture notes using Inkscape]]): type a title, press one key, draw, save, carry on. The keystroke reference is =~/Documents/org/inkscape-figures-cheatsheet.org=.

Linux and X11 only. The module loads on macOS but nothing in it is useful there.

*** install
#+begin_src sh
sudo pacman -S inkscape rofi pdf2svg python-xlib xclip texlive-latexextra
#+end_src

| Requirement                 | Purpose                                                        |
|-----------------------------+----------------------------------------------------------------|
| =inkscape= (1.x)            | drawing, and exporting a figure as PDF plus a LaTeX text layer |
| =pdflatex=, =pdf2svg=       | typeset the labels and turn the result back into an SVG        |
| =standalone=, =transparent= | LaTeX packages used by the wrapper (=texlive-latexextra=)      |
| =python-xlib=               | the shortcut manager reads keys through Xlib                   |
| =xclip=                     | the shortcut manager pastes styles and text via the clipboard  |
| =rofi=                      | the shortcut manager's "save as" prompt for styles and objects |

The shortcut manager is not packaged. Clone it once:

#+begin_src sh
git clone https://github.com/gillescastel/inkscape-shortcut-manager ~/.local/src/clonedStuff/inkscape-shortcut-manager
#+end_src

The figure template lives beside this file in =templates/inkscape-figure.svg= and is edited by hand (open it in Inkscape, change it, save). On a new machine =restow= links the =templates= directory into =~/.config/emacs=.

*** where figures live
Every note has its own folder, so two notes can both have a =graph.svg=. A denote note uses its identifier, =figures/20261114T140132/graph.svg=, which stays the same when the note's title or keywords change. Any other Org file uses its base name, =figures/SICP/graph.svg=; renaming such a file means renaming the folder by hand and fixing the links.

A title that would reuse a name in the same note gets a number instead of a prompt: =graph.svg=, =graph-2.svg=, =graph-3.svg=.
#+begin_src emacs-lisp :tangle "tony-emacs-modules/tony-emacs-inkscape.el" :mkdirp yes
(defun my/inkscape--note-file ()
  "File of the current note, or a `user-error' when there is none."
  (or (buffer-file-name (buffer-base-buffer))
      (user-error "This buffer is not visiting a file")))

(defun my/inkscape--figure-dir (&optional file)
  "Figure directory of note FILE, default the current buffer's file."
  (let* ((file (or file (my/inkscape--note-file)))
         (id (or (and (fboundp 'denote-retrieve-filename-identifier)
                      (ignore-errors (denote-retrieve-filename-identifier file)))
                 (file-name-base file))))

     (expand-file-name
      id (expand-file-name my/inkscape-figures-directory
                           (file-name-directory file))))))

(defun my/inkscape--rendered (file)
  "Path of the LaTeX-rendered copy of figure FILE."
  (expand-file-name (concat ".rendered/" (file-name-nondirectory file))
                    (file-name-directory file)))

(defun my/inkscape--slug (title dir)
  "Turn TITLE into a file name base not yet used by an SVG in DIR."
  (let* ((base (string-trim (replace-regexp-in-string
                             "[^a-z0-9]+" "-" (downcase title))
                            "-+" "-+"))
         (base (if (string-empty-p base) "figure" base))
         (slug base)
         (n 1))
    (while (file-exists-p (expand-file-name (concat slug ".svg") dir))
      (setq slug (format "%s-%d" base (setq n (1+ n)))))
    slug))

(defun my/inkscape--link-block (title file &optional indent)
  "Org text that includes figure FILE with caption TITLE.
Each line is indented by INDENT columns."
  (let ((pad (make-string (or indent 0) ?\s)))
    (format "%s#+CAPTION: %s\n%s#+NAME: fig:%s\n%s[[file:%s]]"
            pad title pad (file-name-base file) pad
            (file-relative-name file))))

*** rendering on save
Inkscape text is not rendered as text. On every save a watcher on the note's figure folder runs three programs in a row, in the background:

1. =inkscape --export-latex= splits the figure into a PDF without text and a =.pdf_tex= file that places each text object with LaTeX;
2. =pdflatex= compiles a small =standalone= document that loads =my/latex-math-macros= and inputs the =.pdf_tex=;
3. =pdf2svg= turns that into =figures/<note>/.rendered/<name>.svg=.

So =$\RR \to f(x)$= typed as plain text in Inkscape shows as real maths in the note, with the same macros as the inline previews. Pure black in the result is replaced by =currentColor=, which Emacs fills with the default foreground: figures are light on a dark theme and follow theme changes. Colours other than black are kept as drawn, including white fills.

The Org link points at the source SVG, never at the rendered copy, so export and other tools see the real figure. =.rendered= can be deleted at any time; =M-x my/inkscape-render-figures= rebuilds it for the current note (useful on a second machine, if the folder is not synced).

A failure, typically a LaTeX error in a label, leaves the previous rendered copy in place, prints a one-line message and puts the details in the =*inkscape-figures*= buffer. Inkscape writes a file more than once per save, so events are collected for half a second before a render starts, and a save that arrives during a render queues exactly one more.
#+begin_src emacs-lisp :tangle "tony-emacs-modules/tony-emacs-inkscape.el" :mkdirp yes
(defvar my/inkscape--watches (make-hash-table :test #'equal)
  "Figure directory -> file notification descriptor.")
(defvar my/inkscape--timers (make-hash-table :test #'equal)
  "Figure file -> pending debounce timer.")
(defvar my/inkscape--renders (make-hash-table :test #'equal)
  "Figure file -> `running', or `dirty' when it changed during the render.")

(defun my/inkscape--watch (dir)
  "Render figures in DIR whenever they are saved."
  (let ((dir (file-name-as-directory (expand-file-name dir))))
    (when (and (file-directory-p dir)
               (not (file-notify-valid-p (gethash dir my/inkscape--watches))))
      (puthash dir (file-notify-add-watch dir '(change) #'my/inkscape--notify)
               my/inkscape--watches))))

(defun my/inkscape--notify (event)
  "Schedule a render for the figure that file notification EVENT is about."
  (pcase-let* ((`(,_ ,action ,file . ,rest) event)
               (file (if (eq action 'renamed) (car rest) file)))
    (when (and (memq action '(created changed renamed))
               (string-suffix-p ".svg" file)
               (not (string-prefix-p "." (file-name-nondirectory file))))
      (when-let* ((timer (gethash file my/inkscape--timers)))
        (cancel-timer timer))
      (puthash file (run-with-timer 0.5 nil #'my/inkscape--render file)
               my/inkscape--timers))))

(defun my/inkscape--log (format-string &rest args)
  "Append a line to the *inkscape-figures* buffer."
  (with-current-buffer (get-buffer-create "*inkscape-figures*")
    (goto-char (point-max))
    (insert (apply #'format format-string args) "\n")))

(defun my/inkscape--render (file)
  "Build the rendered copy of figure FILE asynchronously."
  (remhash file my/inkscape--timers)
  (cond
   ((gethash file my/inkscape--renders)
    (puthash file 'dirty my/inkscape--renders))
   ((file-exists-p file)
    (let* ((name (file-name-base file))
           (dir (file-name-directory (my/inkscape--rendered file))))
      (make-directory dir t)
      (when (zerop (hash-table-count my/inkscape--renders))
        (with-current-buffer (get-buffer-create "*inkscape-figures*")
          (erase-buffer)))
      (puthash file 'running my/inkscape--renders)
      (my/inkscape--log "== %s  %s" (format-time-string "%T") file)
      (delete-file (expand-file-name (concat name ".pdf_tex") dir))
      (with-temp-file (expand-file-name (concat name ".wrap.tex") dir)
        (insert "\\documentclass{standalone}\n"
                "\\usepackage{graphicx,xcolor,transparent,calc,amsmath,amssymb}\n"
                my/latex-math-macros "\n"
                "\\begin{document}\n"
                "\\input{" name ".pdf_tex}\n"
                "\\end{document}\n"))
      (my/inkscape--chain
       file dir
       `(("inkscape" ,file "--export-area-drawing" "--export-type=pdf"
          "--export-latex" ,(concat "--export-filename=" dir name ".pdf"))
         ("pdflatex" "-interaction=nonstopmode" "-halt-on-error"
          ,(concat name ".wrap.tex"))
         ("pdf2svg" ,(concat name ".wrap.pdf") ,(concat name ".tmp.svg"))))))))

(defun my/inkscape--chain (file dir commands)
  "Run COMMANDS one after another in DIR, then finish the render of FILE."
  (if (null commands)
      (my/inkscape--finish file dir)
    (let ((default-directory dir))
      (make-process
       :name "inkscape-figure"
       :buffer (get-buffer-create "*inkscape-figures*")
       :command (car commands)
       :connection-type 'pipe
       :noquery t
       :sentinel
       (lambda (process _event)
         (when (memq (process-status process) '(exit signal))
           (if (zerop (process-exit-status process))
               (my/inkscape--chain file dir (cdr commands))
             (my/inkscape--done file dir)
             (my/inkscape--log "!! %s failed" (caar commands))
             (message "Figure %s: %s failed, see *inkscape-figures*"
                      (file-name-base file) (caar commands)))))))))

(defun my/inkscape--finish (file dir)
  "Install the freshly rendered copy of FILE from DIR and show it."
  (let ((tmp (expand-file-name (concat (file-name-base file) ".tmp.svg") dir)))
    (with-temp-buffer
      (insert-file-contents tmp)
      (while (re-search-forward "rgb(0%, ?0%, ?0%)\\|#000000" nil t)
        (replace-match "currentColor" t t))
      (write-region nil nil tmp nil 'silent))
    (rename-file tmp (my/inkscape--rendered file) t))
  (my/inkscape--done file dir)
  (my/inkscape--refresh file))

(defun my/inkscape--done (file dir)
  "Remove intermediate files for FILE in DIR; render again if it changed."
  (dolist (ext '(".pdf" ".pdf_tex" ".wrap.tex" ".wrap.pdf" ".wrap.aux"
                 ".wrap.log" ".tmp.svg"))
    (delete-file (expand-file-name (concat (file-name-base file) ext) dir)))
  (let ((state (gethash file my/inkscape--renders)))
    (remhash file my/inkscape--renders)
    (when (eq state 'dirty)
      (my/inkscape--render file))))

(defun my/inkscape--refresh (file)
  "Redisplay the preview of every Org link to figure FILE."
  (let ((name (file-name-nondirectory file)))
    (dolist (buffer (buffer-list))
      (with-current-buffer buffer
        (when (derived-mode-p 'org-mode)
          (save-excursion
            (save-restriction
              (widen)
              (goto-char (point-min))
              (while (search-forward name nil t)
                (when-let* ((link (org-element-lineage
                                   (save-excursion (backward-char)
                                                   (org-element-context))
                                   'link t))
                            ((equal (org-element-property :type link) "file"))
                            ((equal (expand-file-name
                                     (org-element-property :path link))
                                    file)))
                  (org-link-preview-region nil t (org-element-begin link)
                                           (org-element-end link)))))))))))

(defun my/inkscape-render-figures ()
  "Render every figure of the current note again."
  (interactive nil org-mode)
  (let* ((dir (my/inkscape--figure-dir))
         (files (and (file-directory-p dir)
                     (directory-files dir t "\\`[^.].*\\.svg\\'"))))
    (mapc #'my/inkscape--render files)
    (message "Rendering %d figure(s)" (length files))))
#+end_src

*** showing the rendered copy
The =file= link type's =:preview= function is wrapped: a link to an SVG that has a rendered copy beside it shows the copy, anything else behaves as before. Link previews are switched on for every Org buffer at startup (=C-c C-x C-v= toggles them in a buffer), and opening a note starts the watcher if the note already has a figure folder.

A new figure shows as an empty box the size of the page until its first save from Inkscape. A figure with no rendered copy (drawn elsewhere, or =.rendered= deleted) is shown raw: black on transparent, hard to see on a dark theme, until it is saved again or =M-x my/inkscape-render-figures= is run.

*** the one key: =C-c e=
=C-c e= (=SPC c e= in meow normal state) in an Org buffer does one of three things, depending on the current line:

| Line                       | What happens                                                                                 |
|----------------------------+----------------------------------------------------------------------------------------------|
| has a link to an =.svg=    | opens that figure in Inkscape                                                                |
| has other text             | the text is the title: creates the figure from the template, replaces the line with caption, name and link, opens Inkscape |
| is empty                   | asks which of this note's figures to open, newest first, and puts its link block on the kill ring |

Opening a figure also starts the watcher for its folder and the shortcut manager. Inkscape is started detached, so it survives an Emacs restart.
#+begin_src emacs-lisp :tangle "tony-emacs-modules/tony-emacs-inkscape.el" :mkdirp yes
(defun my/inkscape--open (file)
  "Edit figure FILE in Inkscape."
  (unless (executable-find "inkscape")
    (user-error "Inkscape is not installed"))
  (my/inkscape--watch (file-name-directory file))
  (my/inkscape-shortcut-manager-start)
  (call-process "inkscape" nil 0 nil file))

(defun my/inkscape--create (title)
  "Replace the current line with a new figure called TITLE and open it."
  (unless (file-exists-p my/inkscape-template)
    (user-error "No figure template at %s" my/inkscape-template))
  (let* ((dir (my/inkscape--figure-dir))
         (file (expand-file-name
                (concat (my/inkscape--slug title dir) ".svg") dir))
         (indent (current-indentation)))
    (make-directory dir t)
    (copy-file my/inkscape-template file)
    (delete-region (pos-bol) (pos-eol))
    (insert (my/inkscape--link-block title file indent))
    (my/inkscape--open file)
    (message "Created %s" (file-relative-name file))))

(defun my/inkscape--pick ()
  "Choose one of the current note's figures, open it and copy its link."
  (let* ((dir (my/inkscape--figure-dir))
         (files (and (file-directory-p dir)
                     (sort (directory-files dir t "\\`[^.].*\\.svg\\'")
                           #'file-newer-than-file-p)))
         (names (mapcar #'file-name-base files)))
    (unless names
      (user-error "This note has no figures yet; type a title on the line first"))
    (let* ((name (completing-read
                  "Figure: "
                  (lambda (string pred action)
                    (if (eq action 'metadata)
                        '(metadata (display-sort-function . identity)
                                   (cycle-sort-function . identity))
                      (complete-with-action action names string pred)))
                  nil t))
           (file (expand-file-name (concat name ".svg") dir)))
      (kill-new (my/inkscape--link-block
                 (let ((title (string-replace "-" " " name)))
                   (concat (upcase (substring title 0 1)) (substring title 1)))
                 file))
      (my/inkscape--open file)
      (message "Opened %s; its link block is on the kill ring" name))))

(defun my/inkscape-figure-dwim ()
  "Open, create or pick an Inkscape figure, depending on the current line.
On a line with a link to an SVG, open it.  On a line with other
text, use the text as the title of a new figure.  On an empty
line, choose among the note's figures."
  (interactive nil org-mode)
  (let ((line (string-trim (buffer-substring-no-properties (pos-bol) (pos-eol)))))
    (cond
     ((string-match "\\[\\[\\(?:file:\\)?\\([^]]+\\.svg\\)\\]" line)
      (my/inkscape--open (expand-file-name (match-string 1 line))))
     ((string-empty-p line) (my/inkscape--pick))
     ((string-match-p "\\`\\(?:\\*+ \\|#\\+\\)" line)
      (user-error "Put the figure title on a plain line of its own"))
     (t (my/inkscape--create line)))))

(with-eval-after-load 'org
  (keymap-set org-mode-map "C-c e" #'my/inkscape-figure-dwim))

Castel's [[https://github.com/gillescastel/inkscape-shortcut-manager][inkscape-shortcut-manager]] grabs the keyboard of every Inkscape window and turns key chords into styles (=f= + =s= fills and strokes the selection, =a= + =d= makes a dotted arrow) and single keys into tools. *** braces: =k= in Inkscape
Inkscape has no brace tool, so a small Inkscape extension, =Extensions > Tony > Brace from line=, turns a line into a curly brace between its two ends. Draw a straight line with the Bezier pen (two clicks, the second a double-click), keep it selected and press =k=. The tip points to the left of the direction the line was drawn: drawn bottom to top it points left, left to right it points up. Press =k= again on the brace to flip it to the other side.

The brace is the classic shape: quarter curls at the ends and at the tip, straight in between. Its height is fixed at 2.5 mm whatever the length, so long and short braces match, and shrinks only on braces too short for it. The line's colour and width are kept; arrowheads are removed. The brace stores its two ends in a =data-brace= attribute, which is how =k= finds them again to flip it.

=k=, =j= (axes) and =y= (parabolas), described below, are free in Inkscape's default keymap and the shortcut manager passes it through. The binding lives in =~/.config/inkscape/keys/default.xml=, a link into the =inkscape= package. Inkscape rewrites it when shortcuts are changed in its preferences; tangling this file again overwrites those changes, so add other bindings here. Inkscape reads extensions and key bindings at startup: restart it after tangling. In the key file the action needs the =app.= prefix (=app.org.tony.brace=), although =inkscape --action-list= prints it without one; without the prefix the key silently does nothing.

The extension and key file are tangled with the languages =python= and =xml= (see =config.py= above for the command).
#+begin_src python :tangle "../../../inkscape/.config/inkscape/extensions/tony_brace.py" :mkdirp yes
#!/usr/bin/env python3
import math

import inkex
from inkex import PathElement

AMPLITUDE = "2.5mm"


class Brace(inkex.EffectExtension):
    def effect(self):
        amplitude = self.svg.viewport_to_unit(AMPLITUDE)
        for node in self.svg.selection.filter(PathElement).values():
            self.brace(node, amplitude)

    def brace(self, node, amplitude):
        ends = node.get("data-brace")
        if ends:
            x1, y1, x0, y0 = map(float, ends.split())
        else:
            points = list(node.path.end_points)
            (x0, y0), (x1, y1) = tuple(points[0]), tuple(points[-1])
        length = math.hypot(x1 - x0, y1 - y0)
        if length == 0:
            return
        t = node.composed_transform()
        scale = math.sqrt(abs(t.a * t.d - t.b * t.c)) or 1
        amp = min(amplitude / scale, length / 3)
        h = amp / 2
        ux, uy = (x1 - x0) / length, (y1 - y0) / length
        # Left of the direction of travel, with SVG's y axis pointing down.
        nx, ny = uy, -ux

        def p(along, out):
            return f"{x0 + ux * along + nx * out:.4f},{y0 + uy * along + ny * out:.4f}"

        m = length / 2
        node.path = (
            f"M {p(0, 0)} Q {p(0, h)} {p(h, h)} L {p(m - h, h)} "
            f"Q {p(m, h)} {p(m, amp)} Q {p(m, h)} {p(m + h, h)} "
            f"L {p(length - h, h)} Q {p(length, h)} {p(length, 0)}"
        )
        node.set("data-brace", f"{x0} {y0} {x1} {y1}")
        node.set("sodipodi:nodetypes", None)
        for key in ("marker-start", "marker-mid", "marker-end"):
            node.style[key] = "none"
        node.style["fill"] = "none"
        node.style["stroke-linejoin"] = "round"
        node.style["stroke-linecap"] = "round"


if __name__ == "__main__":
    Brace().run()

*** axes: =j= in Inkscape
A second extension, =Extensions > Tony > Axis from line=, turns a Bezier line into a number line. With the pen, click one end, single-click where 0 goes, double-click the other end, then press =j=. The line becomes a straight axis between the two ends, with a tick at every grid step out from 0 in both directions, and none at 0 itself. Without the middle click, 0 is at the first end. A middle click off the line is projected onto it.

Ticks are one grid square apart (10 mm in the figure template); with snapping on (=x=) they land on grid points. They carry no numbers, so a tick can stand for 1 or for a million: label the ones that matter with =t=. Setting =LABELS = True= in the script numbers every tick instead (=$n$= text below a horizontal axis and left of a vertical one, counting grid squares, growing rightwards and upwards whichever end was drawn first), typeset by LaTeX on save like any other label. The axis and its ticks (and labels) are grouped: one click selects all of them, a saved colour such as =s r= recolours line and ticks, and =Ctrl+Shift+G= ungroups. The line keeps its style, so an arrowhead goes on with =a= + =s= before =j=; a chord on the finished group also styles the ticks and puts arrows on them. The ticks take the line's colour and width.
#+begin_src python :tangle "../../../inkscape/.config/inkscape/extensions/tony_axis.py" :mkdirp yes
#!/usr/bin/env python3
import math

import inkex
from inkex import Group, PathElement, TextElement

TICK = "1.2mm"
GAP = "1.2mm"
FONT = "3.4mm"
LABELS = False


class Axis(inkex.EffectExtension):
    def effect(self):
        for node in list(self.svg.selection.filter(PathElement).values()):
            self.axis(node)

    def grid_spacing(self):
        grid = self.svg.namedview.findone("inkscape:grid")
        if grid is not None and grid.get("spacingx"):
            return float(grid.get("spacingx"))
        return self.svg.viewport_to_unit("10mm")

    def axis(self, node):
        points = [tuple(p) for p in node.path.end_points]
        if len(points) < 2:
            return
        (sx, sy), (ex, ey) = points[0], points[-1]
        zx, zy = points[1] if len(points) > 2 else points[0]
        length = math.hypot(ex - sx, ey - sy)
        if length == 0:
            return
        t = node.composed_transform()
        scale = math.sqrt(abs(t.a * t.d - t.b * t.c)) or 1
        spacing = self.grid_spacing()
        tick = self.svg.viewport_to_unit(TICK) / scale
        gap = self.svg.viewport_to_unit(GAP) / scale
        font = self.svg.viewport_to_unit(FONT) / scale
        ux, uy = (ex - sx) / length, (ey - sy) / length
        nx, ny = -uy, ux
        zero = (zx - sx) * ux + (zy - sy) * uy
        horizontal = abs(ux) >= abs(uy)

        node.path = f"M {sx:.4f},{sy:.4f} L {ex:.4f},{ey:.4f}"
        node.set("sodipodi:nodetypes", None)

        group = Group()
        node.getparent().insert(node.getparent().index(node), group)
        group.append(node)

        ticks = []
        labels = []
        # Positive values go right on a horizontal axis and up on a vertical
        # one, whichever end was drawn first.
        sign = (1 if ux > 0 else -1) if horizontal else (1 if uy < 0 else -1)
        eps = 1e-6 * length
        below = math.floor(-zero / spacing - 1e-9)
        above = math.ceil((length - zero) / spacing + 1e-9)
        for step in range(below, above + 1):
            along = zero + step * spacing
            if step == 0 or along < -eps or along > length + eps:
                continue
            value = step * sign
            px, py = sx + ux * along, sy + uy * along
            ticks.append(
                f"M {px + nx * tick:.4f},{py + ny * tick:.4f} "
                f"L {px - nx * tick:.4f},{py - ny * tick:.4f}"
            )
            if not LABELS:
                continue
            if horizontal:
                labels.append((px, py + tick + gap + font * 0.75, "middle", value))
            else:
                labels.append((px - tick - gap, py + font * 0.35, "end", value))

        if ticks:
            marks = PathElement()
            marks.path = " ".join(ticks)
            marks.style = {
                "fill": "none",
                "stroke": node.style.get("stroke", "#000000"),
                "stroke-width": node.style.get("stroke-width", "0.5"),
            }
            group.append(marks)
        for x, y, anchor, value in labels:
            text = TextElement(x=f"{x:.4f}", y=f"{y:.4f}")
            text.text = f"${value}$"
            text.style = {
                "font-size": f"{font:.4f}",
                "font-family": "monospace",
                "fill": "#000000",
                "stroke": "none",
                "text-anchor": anchor,
                "text-align": "center" if anchor == "middle" else "end",
            }
            group.append(text)


if __name__ == "__main__":
    Axis().run()
#+end_src

*** parabolas: =y= in Inkscape
=Extensions > Tony > Parabola from line= replaces a pen line with an exact parabola that opens up or down. With two clicks, the first is the vertex and the second a point on one arm; the parabola is drawn symmetrically, as wide on both sides. With three or more clicks, it passes through the first, second and last click, for example two roots and the vertex, and spans from the leftmost to the rightmost of them. No two of the three clicks may be straight above one another; if they are, nothing changes. With snapping on (=x=) the clicks land on grid points, so a parabola through given points comes out exact. The line's colour and width are kept and arrowheads removed.

A parabola segment is exactly one quadratic Bezier curve, so the result is a single smooth segment that Inkscape's node tool can still edit.
#+begin_src python :tangle "../../../inkscape/.config/inkscape/extensions/tony_parabola.py" :mkdirp yes
#!/usr/bin/env python3
import inkex
from inkex import PathElement


class Parabola(inkex.EffectExtension):
    def effect(self):
        for node in self.svg.selection.filter(PathElement).values():
            self.parabola(node)

    def parabola(self, node):
        points = [tuple(p) for p in node.path.end_points]
        if len(points) < 2:
            return
        if len(points) == 2:
            (hx, hy), (px, py) = points
            if px == hx:
                return
            a = (py - hy) / (px - hx) ** 2
            b, c = -2 * a * hx, a * hx * hx + hy
            x0, x1 = hx - abs(px - hx), hx + abs(px - hx)
        else:
            (ax, ay), (bx, by), (cx, cy) = points[0], points[1], points[-1]
            if len({ax, bx, cx}) < 3:
                return
            # Lagrange form of the quadratic through the three points.
            la = ay / ((ax - bx) * (ax - cx))
            lb = by / ((bx - ax) * (bx - cx))
            lc = cy / ((cx - ax) * (cx - bx))
            a = la + lb + lc
            b = -(la * (bx + cx) + lb * (ax + cx) + lc * (ax + bx))
            c = la * bx * cx + lb * ax * cx + lc * ax * bx
            x0, x1 = min(ax, bx, cx), max(ax, bx, cx)

        def f(x):
            return a * x * x + b * x + c

        # A parabola segment is exactly a quadratic Bezier: the control point
        # is where the tangents at the two ends meet.
        xm = (x0 + x1) / 2
        ym = f(x0) + (2 * a * x0 + b) * (xm - x0)
        node.path = f"M {x0:.4f},{f(x0):.4f} Q {xm:.4f},{ym:.4f} {x1:.4f},{f(x1):.4f}"
        node.set("sodipodi:nodetypes", None)
        for key in ("marker-start", "marker-mid", "marker-end"):
            node.style[key] = "none"
        node.style["fill"] = "none"


if __name__ == "__main__":
    Parabola().run()

*** PDF export
Org's LaTeX exporter turns a link to an SVG into =\includesvg=, from the LaTeX =svg= package. That package does at export time what the watcher does on save: it runs Inkscape to split the figure into PDF and text, and LaTeX typesets the text in the document's own font. It needs the package loaded and LaTeX allowed to run programs (=-shell-escape=).

=-shell-escape= lets any document being compiled run any command, so it is not switched on globally. =M-x my/inkscape-org-export-pdf= exports the current buffer with it for that one run, loading =svg= and =my/latex-math-macros= as well, and opens the PDF. =org-latex-pdf-process= itself is left alone, so =C-c C-e l p= still compiles without shell escape and will fail on a file with SVG figures.

The =svg= package keeps its converted files in =svg-inkscape/= beside the exported =.tex=; it is a cache and can be deleted. Set the width per figure with =#+ATTR_LATEX: :width 0.6\linewidth= above the link.

The clone carries one local patch. =t= pastes the typed text inside an SVG snippet without escaping it, so =<= starts a bogus XML tag and Inkscape drops it: =$a < 0$= arrives as =$a  0$=. In =vim.py=, import =escape= from =html= and wrap the text in it (=>{escape(latex)}</tspan>=). ** ~project.el~
Search with globs: =my/consult-ripgrep-globs= asks for include then exclude globs, then runs =consult-ripgrep= so the query comes last. Both prompts start with the previous answer, so repeating a search is =RET RET=. Entries are comma or space separated, and each is normalised:

- A bare word such as =md= or =spec.ts= becomes the extension glob =*.md= or =*.spec.ts=.
- An entry containing =/= is a path glob matched anywhere in the tree, as in VS Code, by prefixing =**/= unless it starts with =/= or =**=. A trailing =/=, =/*= or =/**= is dropped and =/**= is added, so =library-name/*= covers every file under any folder named =library-name=, however deeply nested. Start the entry with =/= to anchor it to the project root.
- Anything else containing =*=, =?= or =[= is used as written.

Include globs are OR'd by ripgrep, so use a path to scope a folder and the exclude prompt to drop file types.

#+begin_src emacs-lisp :tangle "tony-emacs-modules/tony-emacs-project.el" :mkdirp yes
;;; tony-emacs-project.el --- project.el tweaks  -*- lexical-binding: t; -*-
(defvar my/consult-ripgrep-last-include "")
(defvar my/consult-ripgrep-last-exclude "")

(defun my/ripgrep-expand-globs (input prefix)
  "Turn comma or space separated INPUT into ripgrep --glob arguments.
PREFIX is \"\" for includes and \"!\" for excludes."
  (mapcan
   (lambda (tok)
     (let ((path-p (string-match-p "/" tok))
           (tok (replace-regexp-in-string "/\\(\\*\\*?\\)?\\'" "" tok)))
       (cond
        (path-p
         (let ((tok (if (string-match-p "\\`\\(/\\|\\*\\*\\)" tok) tok (concat "**/" tok))))
           (list (concat " --glob=" prefix tok)
                 (concat " --glob=" prefix tok "/**"))))
        ((string-match-p "[*?[]" tok)
         (list (concat " --glob=" prefix tok)))
        (t
         (list (concat " --glob=" prefix "*." (string-remove-prefix "." tok)))))))
   (split-string input "[, ]+" t)))

(defun my/consult-ripgrep-globs (&optional dir)
  "Prompt for include and exclude globs, then run `consult-ripgrep'.
With prefix argument DIR, also prompt for the directory."
  (interactive "P")
  (let* ((inc (read-string "Include globs: " my/consult-ripgrep-last-include))
         (exc (read-string "Exclude globs: " my/consult-ripgrep-last-exclude))
         (consult-ripgrep-args
          (concat consult-ripgrep-args
                  (apply #'concat (my/ripgrep-expand-globs inc ""))
                  (apply #'concat (my/ripgrep-expand-globs exc "!")))))
    (setq my/consult-ripgrep-last-include inc
          my/consult-ripgrep-last-exclude exc)
    (consult-ripgrep dir)))

(use-package project
  :ensure nil
  :bind (:map project-prefix-map
              ("g" . consult-ripgrep)
              ("G" . my/consult-ripgrep-globs))
  :config
  (setq xref-search-program 'ripgrep))
#+end_src

*** copy relative and absolute path with position
Copy a path to the clipboard that VS Code and most terminals/editors jump straight to: ~file:line:column~, with a 1-based character column (a tab counts as one). With an active region it copies ~file:START-END~ lines instead. In dired it copies the path of the file at point with no position. ~C-c y p~ is relative to the project root (or ~default-directory~ with no project); ~C-c y P~ is absolute.
#+begin_src emacs-lisp :tangle "tony-emacs-modules/tony-emacs-tweakers.el" :mkdirp yes
(defun my/path-with-position (absolute)
  "Return the current file path plus position, relative unless ABSOLUTE."
  (let* ((file-path
          (cond
           ((derived-mode-p 'dired-mode) (dired-get-file-for-visit))
           ((buffer-file-name) (buffer-file-name))
           (t (error "No file associated with this buffer"))))
         (root (or (when-let* ((proj (project-current)))
                     (project-root proj))
                   default-directory))
         (path (if absolute
                   (expand-file-name file-path)
                 (file-relative-name file-path root))))
    (cond
     ((derived-mode-p 'dired-mode) path)
     ((use-region-p)
      (format "%s:%d-%d" path
              (line-number-at-pos (region-beginning))
              (line-number-at-pos (region-end))))
     (t
      (format "%s:%d:%d" path
              (line-number-at-pos)
              (1+ (- (point) (line-beginning-position))))))))

(defun my/copy-relative-path-with-line ()
  "Copy the project-relative path with line and column to the clipboard."
  (interactive)
  (let ((result (my/path-with-position nil)))
    (kill-new result)
    (message "Copied: %s" result)))

(defun my/copy-absolute-path-with-line ()
  "Copy the absolute path with line and column to the clipboard."
  (interactive)
  (let ((result (my/path-with-position t)))
    (kill-new result)
    (message "Copied: %s" result)))

(global-set-key (kbd "C-c y p") #'my/copy-relative-path-with-line)
(global-set-key (kbd "C-c y P") #'my/copy-absolute-path-with-line)

*** Naysayer

Based on theme from Jonathan Blow's livestreams.

#+begin_src emacs-lisp :tangle "tony-emacs-modules/tony-emacs-themes.el" :mkdirp yes
(defun tony/lexbind-cookie-add-to-file (file)
  "Add a `lexical-binding' cookie to FILE's first line if it lacks one.
Idempotent: does nothing if the cookie is already present. Note this dirties
the package's git worktree if FILE lives under elpaca/sources; if
`elpaca-update' ever balks, run `git checkout -- .' in that source dir."
  (when (file-exists-p file)
    (with-temp-buffer
      (insert-file-contents file)
      (goto-char (point-min))
      (unless (search-forward "lexical-binding" (line-end-position) t)
        (end-of-line)
        (insert "  -*- lexical-binding: t; -*-")
        (write-region (point-min) (point-max) file nil 'quiet)))))

(defun tony/add-lexbind-cookie (e)
  "Elpaca `:build' step: patch E's main file via `tony/lexbind-cookie-add-to-file'.
For third-party packages that ship without a `lexical-binding' cookie, so
Emacs 31 warns on every load. Only runs on a package's first build --
`elpaca<-builtp' is just `(file-exists-p build-dir)', so once a package has
been built once, every later startup takes the two-step
`elpaca--pre-built-steps' path and skips `:build' forever. See
`tony/ensure-lexbind-cookie' for the self-healing counterpart that covers
that case."
  (when-let* ((dir (elpaca<-source-dir e))
              (file (expand-file-name (format "%s.el" (elpaca<-package e)) dir)))
    (tony/lexbind-cookie-add-to-file file))
  (elpaca-continue e))

(defun tony/ensure-lexbind-cookie (package)
  "Patch PACKAGE's built file via `tony/lexbind-cookie-add-to-file', if built.
Meant for a `use-package' `:init' form, which -- unlike elpaca's `:build'
steps -- always runs at startup regardless of whether the package was
already built in a previous session. Belt-and-suspenders alongside
`tony/add-lexbind-cookie': fixes the common case where the build directory
already existed before this fix was added, with no manual rebuild needed."
  (tony/lexbind-cookie-add-to-file
   (expand-file-name (format "%s/%s.el" package package) elpaca-builds-directory)))

;; Upstream naysayer has no `lexical-binding' cookie (dormant since 2025-05)
;; and Emacs 31 warns on every load, so patch it as a build step, and again
;; (self-healing, in case the build step above was skipped) in :init.
(use-package
 naysayer-theme
 :ensure
 (:build (:before elpaca-build-link tony/add-lexbind-cookie))
 :defer t
 :init
 (tony/ensure-lexbind-cookie "naysayer-theme")

** ~dirvish.el~
[[https://github.com/alexluigit/dirvish][Dirvish]] is a modern file manager built on =dired=. =dirvish-override-dired-mode= makes every =dired= call (=C-x d=, =C-x p D=, =dired-jump=) open Dirvish, so existing habits keep working. It adds file previews, a parent-directory column, subtree expansion (=TAB=), VC state and a quick-access menu (=a=). The icons come from =nerd-icons= through the =nerd-icons= attribute, which replaces =nerd-icons-dired=.

Inside Dirvish, =h= goes up to the parent directory and =l= enters the folder or opens the file at point (=^= and =RET= still work). Meow leaves these free because Dired buffers are in motion state, where only =j= and =k= are claimed. =h= and =l= replace =describe-mode= and =dired-do-redisplay=, still available as =C-h m= and =g=.

Binds: =C-c d= is =dirvish-dwim= (a split with preview, scoped to the current directory) and =C-c D= toggles =dirvish-side=, a project sidebar that follows the current file. Inside Dirvish, =f= opens the =dirvish-fd= file search menu, =y= the yank menu, =N= the narrowing menu, =M-t= toggles the layout and =?= lists everything.

Dirvish works without any external tools, but each tool below unlocks one feature:

| Tool            | Feature                                        |
|-----------------+------------------------------------------------|
| =fd=            | =dirvish-fd= live file search                  |
| =coreutils=     | GNU =ls= (=gls=), needed for =--group-directories-first= on macOS |
| =poppler=       | PDF preview (=pdftoppm=)                       |
| =imagemagick=   | image preview and EXIF                         |
| =ffmpegthumbnailer= | video thumbnails                           |
| =mediainfo=     | audio and video metadata                       |
| =tar=, =unzip=  | archive preview (already on both systems)      |

On macOS the system =ls= is BSD and does not accept the GNU flags Dired uses, so the config switches =insert-directory-program= to =gls= when it is installed. The check lives in =:config= because =gls= is only on =exec-path= once =exec-path-from-shell= has run; Elpaca defers that until =after-init=, and an Emacs launched from the Dock starts with a =PATH= that does not include =/opt/homebrew/bin=. If the check runs too early Dired falls back to BSD =ls=, every listing fails, and Dirvish reports =wrong-type-argument stringp nil=:

#+begin_src sh
brew install coreutils fd poppler ffmpegthumbnailer mediainfo imagemagick
#+end_src

On Arch Linux, where GNU =ls= is already the default:

#+begin_src sh
sudo pacman -S fd poppler ffmpegthumbnailer mediainfo imagemagick
#+end_src

;; agent-shell draws its header SVG at the frame's own font metrics --
;; `frame-char-height', plus the default face's family and size -- so it
;; already tracks the fontaine preset. Emacs then inflates the finished
;; image a second time by `image-scaling-factor', whose `auto' value works
;; out to ~1.6 at this font size, landing the header ~60% larger than body
;; text. An explicit :scale in the image spec beats `image-scaling-factor',
;; and the header string carries exactly one image, so pin it there rather
;; than changing image scaling for the whole buffer.
(defun my/agent-shell-header-unscale (header)
  "Pin the SVG image in HEADER to :scale 1 so it matches body text."
  (when (stringp header)

        (let ((image (get-text-property pos 'display header)))
          (when (imagep image)
            (setf (image-property image :scale) 1)))
        (setq pos (next-single-property-change pos 'display header)))))
  header)

*** render mermaid blocks in org
=mermaid-mode= also defines the Babel executor for mermaid source blocks, which calls the =mmdc= CLI and writes an SVG; it loads with Org so the executor exists before any block has been edited. Do not also install =ob-mermaid=: it defines the same function, and whichever package loads last silently replaces the other. =C-c C-c= renders the block and previews the image under it; =C-c '= edits it in =mermaid-mode=. A named block needs no =:file=: the SVG is called after the =#+name= and goes in the note's Inkscape figure directory (=figures/<denote id>/=, see =inkscape.el=), so drawn figures and diagrams sit together. An unnamed block needs an explicit =:file=, which is still placed in that directory.

#+begin_src org
,#+name: login-flow
,#+begin_src mermaid
graph TD
  A[Login] --> B{Valid?}
  B -->|yes| C[Home]
  B -->|no| A
,#+end_src
#+end_src

The diagram theme is chosen when the block runs: =dark= under a dark Emacs theme, =default= under a light one, on a transparent background. A diagram keeps the colours it was rendered with, so after switching themes re-run it with =C-c C-c=, or the whole buffer with =org-babel-execute-buffer=.

Emacs draws SVG with librsvg, which skips the HTML labels Mermaid uses by default, so flowchart nodes would render as empty boxes. =my/mermaid-config= turns HTML labels off so every label is plain SVG text, and turns off =useMaxWidth= so each SVG carries its real width instead of =100%=. It is written to a JSON file in the temp directory when =mermaid-mode= loads and passed to every =mmdc= run through =mermaid-flags=. Diagrams rendered before this need a re-run.

Org limits a previewed image to =fill-column= by default, which shrinks a wide diagram to unreadable text, and =visual-fill-column-width= does not change it. =org-image-max-width= is set to the window width instead. Org measures whichever window is selected when it draws a preview, so previews are drawn in a window showing the buffer; otherwise a preview drawn from another frame overflows and is cut off.

librsvg also drops the space at the start of each word's =tspan=, so multi-word labels run together. After each render the SVG's root element gets =xml:space="preserve"=, which keeps them.

| Requirement     | Purpose                                     |
|-----------------+---------------------------------------------|
| =mmdc=          | renders a diagram to SVG                    |
| =node=          | runs =mmdc=                                 |
| Chrome/Chromium | =mmdc= renders through headless Puppeteer   |

Puppeteer looks for its own downloaded browser, which neither package below provides on macOS, so =PUPPETEER_EXECUTABLE_PATH= is pointed at the installed Google Chrome or Chromium unless it is already set. On macOS:

#+begin_src sh
brew install mermaid-cli
brew install --cask google-chrome
#+end_src

On Arch Linux the package pulls in =chromium= and =nodejs=:

#+begin_src sh
sudo pacman -S mermaid-cli
#+end_src

On Artix, if =pacman -Si mermaid-cli= finds nothing, enable the Arch =[extra]= repository with =artix-archlinux-support=, or install from npm against the system Chromium:

#+begin_src sh
sudo pacman -S nodejs npm chromium
sudo PUPPETEER_SKIP_DOWNLOAD=1 npm install -g @mermaid-js/mermaid-cli
#+end_src

*** view a diagram at full size
An inline preview cannot be scrolled or panned, so a wide diagram is shown scaled to the window. =C-c C-o= on its result link opens the SVG in =image-mode=. Figures and diagrams under a note's =figures/= directory open at their original size instead of being scaled down to the window, and dragging with the left mouse button pans them; arrow keys and a trackpad swipe also scroll, =i += and =i -= zoom, and =s w= fits the width.
#+begin_src emacs-lisp :tangle "tony-emacs-modules/tony-emacs-mermaid.el" :mkdirp yes
(declare-function image-display-size "image.c")
(declare-function image-get-display-property "image-mode")
(declare-function image-set-window-hscroll "image-mode")
(declare-function image-set-window-vscroll "image-mode")
(declare-function image-transform-original "image-mode")

(defun my/image-drag-pan (event)
  "Pan the image in the window of EVENT by dragging the mouse."
  (interactive "e")
  (with-selected-window (posn-window (event-start event))
    (let* ((size (image-display-size (image-get-display-property) t))
           (char (frame-char-width))
           (max-x (max 0 (- (car size) (window-body-width nil t))))
           (max-y (max 0 (- (cdr size) (window-body-height nil t))))
           (start (cdr (mouse-pixel-position)))
           (x0 (* (window-hscroll) char))
           (y0 (window-vscroll nil t)))
      (track-mouse
        (while (mouse-movement-p (read-event))
          (let ((pos (cdr (mouse-pixel-position))))
            (when (car pos)
              (image-set-window-hscroll
               (/ (min max-x (max 0 (- x0 (- (car pos) (car start))))) char))
              (image-set-window-vscroll
               (min max-y (max 0 (- y0 (- (cdr pos) (cdr start)))))))))))))

(defun my/image-original-size-for-figures ()
  "Show an SVG from a note's figure directory at its original size."
  (when (and buffer-file-name
             (string-match-p "/figures/[^/]+/[^/]+\\.svg\\'" buffer-file-name))
    (image-transform-original)))

(with-eval-after-load 'image-mode
  (keymap-set image-mode-map "<down-mouse-1>" #'my/image-drag-pan)
  (add-hook 'image-mode-hook #'my/image-original-size-for-figures))
#+end_src
